\newglossaryentry{sesgos}{%
	name={sesgos},
	description={El sesgo en inteligencia artificial son las tendencias o prejuicios sistemáticos que pueden influir en el comportamiento o decisiones del modelo de IA}
}

\newglossaryentry{chatbots}{%
	name={chatbots},
	description={Programa informático que usa IA y NLP para comprender las preguntas de los clientes y automatizar las respuestas simulando una conversación humana}
}

\newglossaryentry{bytecode}{%
	name={bytecode},
	description={Es un lenguaje intermedio un poco más abstracto que el lenguaje máquina. En caso de Java, este bytecode será interpretado por la máquina virtual JVM}
}

\newglossaryentry{Web Scraping}{%
	name={Web Scraping},
	description={Es una técnica que se emplea para extraer datos de una página web de forma automatizada, descargando los documentos HTML de una web y extrayendo la información}
}

\newglossaryentry{pipelines}{%
	name={pipelines},
	description={En el entorno de los proyecto de IA es el conjunto de herramientas y proceso orientados a recopilar grandes cantidades de datos, procesarlos, transformarlos y almacenarlos de forma eficiente}
}
